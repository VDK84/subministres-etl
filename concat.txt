---
name: Build, Scan & Publish Docker Image

on:
  push:
    branches:
      - main
  workflow_dispatch:

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: vdk84/subministres-etl

permissions:
  contents: write
  packages: write
  security-events: write

jobs:
  build-and-push:
    name: Build, Scan & Push
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Calculate Semantic Version & Bump Tag
        id: tag_version
        uses: mathieudutour/github-tag-action@v6.2
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          default_bump: patch
          fetch_all_tags: true
          release_branches: main

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Extract Docker metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}
          tags: |
            type=raw,value=latest
            type=raw,value=${{ steps.tag_version.outputs.new_tag }}
            type=raw,value=${{ steps.tag_version.outputs.new_version }}
            type=sha,format=short

      - name: Log in to GitHub Container Registry
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Build Docker image (local load for scan)
        uses: docker/build-push-action@v6
        with:
          context: .
          load: true
          build-args: |
            APP_VERSION=${{ steps.tag_version.outputs.new_version }}
          tags: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:test-scan
          cache-from: type=gha
          cache-to: type=gha,mode=max

      - name: DevSecOps Vulnerability Scan with Trivy
        uses: aquasecurity/trivy-action@v0.36.0
        with:
          image-ref: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:test-scan
          format: "sarif"
          output: "trivy-results.sarif"
          severity: "CRITICAL,HIGH"

      - name: Upload Trivy SARIF report to GitHub Security
        uses: github/codeql-action/upload-sarif@v3
        if: always() && hashFiles('trivy-results.sarif') != ''
        with:
          sarif_file: "trivy-results.sarif"
          category: "trivy-container-scan"

      - name: Build and Push Docker image to GHCR
        uses: docker/build-push-action@v6
        with:
          context: .
          push: true
          build-args: |
            APP_VERSION=${{ steps.tag_version.outputs.new_version }}
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        if: steps.tag_version.outputs.new_tag != ''
        with:
          tag_name: ${{ steps.tag_version.outputs.new_tag }}
          name: Release ${{ steps.tag_version.outputs.new_tag }}
          body: ${{ steps.tag_version.outputs.changelog }}

# pyright: reportMissingImports=false
import logging
from abc import ABC, abstractmethod
from collections.abc import Callable
from datetime import datetime, timedelta
from typing import Any

from src.config import settings
from src.email_client import ImapMailboxClient
from src.influx_writer import influx_writer
from src.models import FacturaMetriques, FacturaResult, SubministreType
from src.storage import storage

logger = logging.getLogger("connector.base")


class BaseConnector(ABC):
    """Classe base abstracta per als connectors de subministraments."""

    name: str = "base"
    tipus: SubministreType = SubministreType.ELECTRICITAT

    def __init__(self) -> None:
        self.logger = logging.getLogger(f"connector.{self.name}")
        self.consecutive_failures: int = 0
        self.cooldown_until: datetime | None = None
        self.progress_callback: Callable[[dict[str, Any]], None] | None = None

    def report_progress(
        self,
        step: str,
        current: int = 0,
        total: int = 0,
        status: str = "running",
    ) -> None:
        """Emet l'estat i el progrés exacte (ex: 3/40) cap a la UI per WebSocket i al log."""
        pct = int((current / total) * 100) if total > 0 else 0
        counter_str = f"[{current}/{total}] " if total > 0 else ""
        self.logger.info(f"[{self.name.upper()}] {counter_str}{step}")
        if self.progress_callback:
            try:
                self.progress_callback(
                    {
                        "connector": self.name.upper(),
                        "step": step,
                        "current": current,
                        "total": total,
                        "percent": pct,
                        "status": status,
                    }
                )
            except (TypeError, ValueError, RuntimeError, OSError) as e:
                self.logger.debug(f"Error progress callback: {e}")

    def is_in_cooldown(self) -> bool:
        """Comprova si el connector està en pausa preventiva de seguretat."""
        if self.cooldown_until and datetime.now() < self.cooldown_until:
            minuts_restants = int((self.cooldown_until - datetime.now()).total_seconds() / 60) + 1
            self.logger.warning(
                f"⛔ PROTECCIÓ DE SEGURETAT ({self.name.upper()}): S'han assolit {self.consecutive_failures} intents fallits. "
                f"Operació cancel·lada durant {minuts_restants} minuts per protegir el compte d'un bloqueig."
            )
            return True
        return False

    def record_failure(self, reason: str) -> None:
        """Registra un error consecutiu i activa el bloqueig preventiu si cal."""
        self.consecutive_failures += 1
        self.logger.warning(

    def _fetch_otp_code_from_email(
        self,
        imap_client: ImapMailboxClient,
        timeout_seconds: int = 75,
    ) -> str | None:
        """Cerca el codi 2FA de 6 dígits enviat per Bonpreu a la bústia IMAP."""
        start_time = time.time()
        self.logger.info(f"Esperant arribada del correu 2FA de Bonpreu (fins a {timeout_seconds}s)...")
        while time.time() - start_time < timeout_seconds:
            emails = imap_client.fetch_recent_emails(
                sender_filters=["bonpreu", "info@bonpreu.cat"],
                days_back=1,
            )
            # Ordenar els correus del més recent al més antic per agafar sempre l'últim codi generat
            emails.sort(key=lambda x: x.date.timestamp() if x.date else 0.0, reverse=True)
            for em in emails:
                subject_clean = em.subject.lower()
                body_clean = em.body_text
                # Comprovar si és el correu de verificació d'identitat
                if "identitat" in subject_clean or "verificaci" in subject_clean or "código" in subject_clean:
                    match = re.search(r"\b(\d{6})\b", body_clean)
                    if match:
                        code = match.group(1)
                        self.logger.info(f"✅ Codi 2FA de Bonpreu capturat del correu: {code}")
                        return code
            time.sleep(3)
        return None

    def extract_metrics_from_pdf(self, pdf_bytes: bytes) -> FacturaMetriques | None:
        """Extreu mètriques detallades del PDF oficial de Bonpreu Esclat Energia."""
        text = extract_text_from_pdf_bytes(pdf_bytes)
        if not text:
            return None

        # 1. Número de factura (ex: Número de la factura:\nF26612441 o 2026E123456)
        num_factura = extract_regex_first(
            text, r"(?:N[úu]mero\s+(?:de\s+la\s+)?factura:?|Factura\s+n[úu]m\.?:?)\s*\n?\s*([A-Za-z0-9/_-]+)"
        )
        if not num_factura or num_factura.lower() == "ero":
            num_factura = extract_regex_first(text, r"(F\d{7,12})")
        if not num_factura or num_factura.lower() == "ero":
            num_factura = extract_regex_first(text, r"Factura\s+([0-9]{4}E[0-9]+)")

        # 2. Data d'emissió
        data_str = extract_regex_first(
            text, r"(?:Data\s*(?:d\x27)?emissi[óo]|Fecha\s*emisi[óo]n):?\s*\n?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})"
        )
        data_emissio = parse_date_catalan_spanish(data_str) if data_str else None

        # 3. Import total
        total_eur = extract_float_value(
            text,
            r"(?:Import\s+total\s+de\s+la\s+factura|Total\s*factura|Import\s*total|Total\s*a\s*pagar)[^\d\n]{0,30}\n?\s*(\d{1,4}(?:[.,]\d{1,2})?)\s*€",
        )
        if total_eur is None:
            total_eur = extract_float_value(
                text, r"(?:Total\s*factura|Import\s*total|Total\s*a\s*pagar):?\s*(\d+(?:[.,]\d{1,2})?)\s*€?"
            )
        if total_eur is None:
            total_eur = parse_euro_amount(text)

        if total_eur is None:
            return None

        # 4. Consums d'electricitat en kWh (Total i per períodes)
        consum_total = extract_float_value(
            text, r"(?:Consum\s*total|Energia\s*consumida|Total\s*consum):?\s*(\d+(?:[.,]\d{1,2})?)\s*kWh"
        )
        p1 = extract_float_value(text, r"(?:Punta|P1)[:\s]+(\d+(?:[.,]\d{1,2})?)\s*kWh")
        p2 = extract_float_value(text, r"(?:Pla|Llano|P2)[:\s]+(\d+(?:[.,]\d{1,2})?)\s*kWh")
        p3 = extract_float_value(text, r"(?:Vall|Valle|P3)[:\s]+(\d+(?:[.,]\d{1,2})?)\s*kWh")

        if consum_total is None and (p1 or p2 or p3):
            consum_total = (p1 or 0.0) + (p2 or 0.0) + (p3 or 0.0)

    def extract_metrics_from_pdf(self, pdf_bytes: bytes) -> FacturaMetriques | None:
        """Parseja el PDF de Catgas per obtenir dades de consum en kWh/m3 i import."""
        text = extract_text_from_pdf_bytes(pdf_bytes)
        if not text:
            return None

        # 1. Número de factura (ex: Factura núm: FG2026/001234)
        num_factura = extract_regex_first(
            text, r"(?:Factura\s*n[úu]m\.?|N[úu]mero\s*de\s*factura:?)\s*([A-Za-z0-9/_-]+)"
        )
        if not num_factura:
            num_factura = extract_regex_first(text, r"(FG\d{4}[/-]?\d+)")

        # 2. Data d'emissió
        data_str = extract_regex_first(
            text, r"(?:Data\s*emissi[óo]|Fecha\s*emisi[óo]n):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})"
        )
        data_emissio = parse_date_catalan_spanish(data_str) if data_str else None

        # 3. Import total
        total_eur = extract_float_value(
            text, r"(?:Total\s*factura|Import\s*total|Total\s*a\s*pagar):?\s*(\d+(?:[.,]\d{1,2})?)\s*€?"
        )
        if total_eur is None:
            total_eur = parse_euro_amount(text)

        if total_eur is None:
            return None

        # 4. Consum de gas en kWh i m3
        consum_kwh = extract_float_value(
            text, r"(?:Consum\s*facturat|Consumo|Energia\s*facturada):?\s*(\d+(?:[.,]\d{1,2})?)\s*kWh"
        )
        if consum_kwh is None:
            consum_kwh = extract_float_value(text, r"(\d+(?:[.,]\d{1,2})?)\s*kWh")

        consum_m3 = extract_float_value(text, r"(\d+(?:[.,]\d{1,2})?)\s*m[3³]")

        # 5. Dates del període
        dt_inici, dt_fi = extract_date_range(
            text,
            r"(?:Per[íi]ode\s*de\s*facturaci[óo]|Per[íi]ode):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})\s*(?:fins\s*a|a|-)\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})",
        )

        # 6. CUPS de gas (ES02...)
        cups = extract_regex_first(text, r"(ES\d{16,20}[A-Za-z]{0,2})")

        return FacturaMetriques(
            proveidor=self.name,
            tipus=self.tipus,
            numero_factura=num_factura or f"CATGAS-{date.today().isoformat()}",
            data_emissio=data_emissio or date.today(),
            periode_inici=dt_inici,
            periode_fi=dt_fi,
            dies_facturats=(dt_fi - dt_inici).days if (dt_inici and dt_fi) else None,
            import_total=total_eur,
            consum_kwh=consum_kwh,
            consum_m3=consum_m3,
            cups_o_referencia=cups,
        )

    def _fallback_metrics(self, pdf_bytes: bytes, fallback_date: date) -> FacturaMetriques | None:
        amount = parse_euro_amount(extract_text_from_pdf_bytes(pdf_bytes))
        if amount is None:
            return None
        return FacturaMetriques(

    def extract_metrics_from_pdf(self, pdf_bytes: bytes) -> FacturaMetriques | None:
        """Parseja el PDF de factura de Digi Mobil."""
        text = extract_text_from_pdf_bytes(pdf_bytes)
        if not text:
            return None

        # 1. Número de factura (ex: FACTURA Número: DGFC2210546467)
        num_factura = extract_regex_first(
            text, r"(?:FACTURA\s+N[úu]mero|N[úu]mero\s*de\s*factura|Factura\s*n[úu]m\.?):\s*([A-Za-z0-9/_-]+)"
        )
        if not num_factura:
            num_factura = extract_regex_first(text, r"(DGFC\d{6,15})")
        if not num_factura:
            num_factura = extract_regex_first(text, r"Factura\s+([A-Za-z0-9/_-]{6,20})")

        # 2. Data d'emissió
        data_str = extract_regex_first(
            text, r"(?:Fecha\s*de\s*emisi[óo]n|Data\s*d'emissi[óo]):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})"
        )
        data_emissio = parse_date_catalan_spanish(data_str) if data_str else None
        if not data_emissio and num_factura:
            # Deducim l'any i el mes del patró oficial DGFC + AA + MM
            dgfc_match = re.search(r"DGFC(\d{2})(\d{2})", num_factura)
            if dgfc_match:
                try:
                    any_d = 2000 + int(dgfc_match.group(1))
                    mes_d = int(dgfc_match.group(2))
                    data_emissio = date(any_d, mes_d, 1)
                except ValueError:
                    pass

        # 3. Import total (prioritzant el total oficial de factura amb 2 decimals)
        total_eur = extract_float_value(
            text,
            r"(?:Total\s*a\s*pagar|TOTAL\s*FACTURA|Import\s*total|Importe\s*total)[^0-9\n]{0,30}?[:\s]\s*(\d{1,4}[.,]\d{2})\s*€?",
        )
        if total_eur is None:
            total_eur = extract_float_value(text, r"(\d{1,4}[.,]\d{2})\s*€?\s*TOTAL\s*FACTURA")
        if total_eur is None:
            total_eur = extract_float_value(text, r"TOTAL\s*SERVICIOS[^0-9]{0,20}?(\d{1,4}[.,]\d{2})\s*€?")
        if total_eur is None:
            total_eur = parse_euro_amount(text)

        if total_eur is None:
            return None

        # 4. Període facturat
        dt_inici, dt_fi = extract_date_range(
            text,
            r"(?:Per[íi]odo\s*de\s*facturaci[óo]n|Per[íi]ode):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})\s*(?:al|a|-)\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})",
        )

        # 5. Dades addicionals de telefonia
        total_gb = extract_float_value(text, r"(\d+(?:[.,]\d{1,2})?)\s*GB")

        return FacturaMetriques(
            proveidor=self.name,
            tipus=self.tipus,
            numero_factura=num_factura or f"DIGI-{date.today().isoformat()}",
            data_emissio=data_emissio or date.today(),
            periode_inici=dt_inici,
            periode_fi=dt_fi,
            dies_facturats=(dt_fi - dt_inici).days if (dt_inici and dt_fi) else None,
            import_total=float(total_eur),
            total_gb_consumits=total_gb,
        )

    def _fetch_otp_code_from_email(
        self,
        imap_client: ImapMailboxClient,
        timeout_seconds: int = 75,
    ) -> str | None:
        """Cerca el codi 2FA de 6 dígits enviat per Veolia a la bústia IMAP."""
        import time

        start_time = time.time()
        self.logger.info(f"Esperant arribada del correu 2FA de Veolia (fins a {timeout_seconds}s)...")
        while time.time() - start_time < timeout_seconds:
            emails = imap_client.fetch_recent_emails(
                sender_filters=list(self.EMAIL_SENDERS),
                days_back=1,
            )
            for em in emails:
                subject_clean = em.subject.lower()
                body_clean = em.body_text
                if (
                    "código" in subject_clean
                    or "codi" in subject_clean
                    or "autentic" in subject_clean
                    or "identitat" in subject_clean
                ):
                    match = re.search(r"\b(\d{6})\b", body_clean)
                    if match:
                        code = match.group(1)
                        self.logger.info(f"✅ Codi 2FA de Veolia capturat del correu: {code}")
                        return code
            time.sleep(3)
        return None

    def extract_metrics_from_pdf(self, pdf_bytes: bytes) -> FacturaMetriques | None:
        """Parseja el PDF de Veolia per obtenir dades estructurades de consum i facturació."""
        text = extract_text_from_pdf_bytes(pdf_bytes)
        if not text:
            return None

        # 1. Detecció de Període Trimestral (Veolia / Agbar facturem per trimestres, ex: 2026/3 o 3 / 2026)
        quarter: int | None = None
        quarter_year: int | None = None

        quarter_match = re.search(
            r"(?:Per[íi]ode(?:[\s\w]*?):?[\s]*)(?:(\d{4})\s*[/_-]\s*([1-4])|([1-4])\s*[/_-]\s*(\d{4})|([1-4])[ººrrtT]?\s*(?:trimestre|trim\.?)\s*(?:de(?:l)?)?\s*(\d{4}))",
            text,
            re.IGNORECASE,
        )
        if quarter_match:
            g1, g2, g3, g4, g5, g6 = quarter_match.groups()
            if g1 and g2:
                quarter_year, quarter = int(g1), int(g2)
            elif g3 and g4:
                quarter, quarter_year = int(g3), int(g4)
            elif g5 and g6:
                quarter, quarter_year = int(g5), int(g6)
        else:
            # Cerca genèrica de patró trimestre/any (ex: "3 / 2026" o "2026/3")
            simple_q = re.search(r"\b([1-4])\s*/\s*(\d{4})\b|\b(\d{4})\s*/\s*([1-4])\b", text)
            if simple_q:
                q_a, y_a, y_b, q_b = simple_q.groups()
                if q_a and y_a and 2000 <= int(y_a) <= 2100:
                    quarter, quarter_year = int(q_a), int(y_a)
                elif y_b and q_b and 2000 <= int(y_b) <= 2100:
                    quarter_year, quarter = int(y_b), int(q_b)

        # 2. Número de factura
        # Exemples: "Número factura: 2026/123456", "Factura núm. B12345678", "Núm. 22272026A100094695"
        num_factura = extract_regex_first(
            text,
            r"(?:Factura\s*(?:n[úu]m\.?|n[ºº])|N[úu]mero\s*(?:de\s*)?factura|N[úu]m\.?\s*(?:de\s*)?factura|N[úu]m\.?)\s*:?\s*([A-Za-z0-9/_-]{6,30})",
        )
        if not num_factura:
            num_factura = extract_regex_first(text, r"Factura\s+([A-Za-z0-9/_-]{6,25})")

        # 3. Data d'emissió
        data_str = extract_regex_first(
            text,
            r"(?:Data\s*(?:d'emissi[óo]|emisi[óo]n|factura|de\s*la\s*factura)?|Fecha\s*(?:de\s*)?emisi[óo]n?):?\s*(\d{1,2}(?:\s*(?:de|d'|/|-)\s*|\s+)[a-zA-ZÀ-ÿ0-9.]+(?:\s*(?:de|del|/|-)\s*|\s+)\d{2,4})",
        )
        if not data_str:
            data_str = extract_regex_first(text, r"(?:Data|Fecha):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})")
        data_emissio = parse_date_catalan_spanish(data_str) if data_str else None

        # 4. Import total
        total_eur = extract_float_value(
            text,
            r"(?:Total\s*(?:factura|a\s*pagar|rebut)?|Import\s*total|Total|Importe\s*total):?\s*(\d+(?:[.,]\d{1,2})?)\s*€?",
        )
        if total_eur is None:
            total_eur = parse_euro_amount(text)

        # Si no tenim com a mínim import total, no és una factura vàlida
        if total_eur is None:
            return None

        # 5. Consum d'aigua en m3
        # Exemples: "Consum facturat: 12 m3", "Consum del període: 14 m³", "Consum: 12 m3"
        consum_m3 = extract_float_value(
            text,
            r"(?:Consum\s*(?:facturat|del\s*per[íi]ode|total)?|Consumo\s*(?:facturado)?):?\s*(\d+(?:[.,]\d{1,2})?)\s*m[3³]",
        )
        if consum_m3 is None:
            consum_m3 = extract_float_value(text, r"(\d+(?:[.,]\d{1,2})?)\s*m[3³]")

        # 6. Dates del període
        dt_inici, dt_fi = extract_date_range(
            text,
            r"(?:Per[íi]ode\s*facturat|Per[íi]odo):?\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})\s*(?:a|fins\s*a|-)\s*(\d{1,2}[/-]\d{1,2}[/-]\d{2,4})",
        )

        # Si no hem trobat rang de dates explícit per regex, però tenim el trimestre (ex: 2026/3):
        if (not dt_inici or not dt_fi) and quarter and quarter_year:
            quarter_ranges = {
                1: (date(quarter_year, 1, 1), date(quarter_year, 3, 31)),
                2: (date(quarter_year, 4, 1), date(quarter_year, 6, 30)),
                3: (date(quarter_year, 7, 1), date(quarter_year, 9, 30)),
                4: (date(quarter_year, 10, 1), date(quarter_year, 12, 31)),
            }
            if quarter in quarter_ranges:
                dt_inici, dt_fi = quarter_ranges[quarter]

        # Si no tenim data d'emissió però tenim trimestre, establim la fi del trimestre com a referència
        if not data_emissio:
            if dt_fi:
                data_emissio = dt_fi
            else:
                data_emissio = date.today()

        # 7. Referència o contracte
        contracte = extract_regex_first(text, r"(?:Contracte|Contrato|P[òo]lissa):?\s*([0-9A-Za-z/_-]+)")

        dies = (dt_fi - dt_inici).days + 1 if (dt_inici and dt_fi) else None

        tags_extra: dict[str, str] = {}
        fields_extra: dict[str, float] = {}
        if quarter:
            tags_extra["trimestre"] = f"T{quarter}"
            if quarter_year:
                tags_extra["periode_trimestral"] = f"{quarter_year}/T{quarter}"
            fields_extra["trimestre_num"] = float(quarter)

        return FacturaMetriques(
            proveidor=self.name,
            tipus=self.tipus,
            numero_factura=num_factura
            or (
                f"VEOLIA-{quarter_year}T{quarter}"
                if (quarter and quarter_year)
                else f"VEOLIA-{date.today().isoformat()}"
            ),
            data_emissio=data_emissio,
            periode_inici=dt_inici,
            periode_fi=dt_fi,
            dies_facturats=dies,
            import_total=float(total_eur),
            consum_m3=consum_m3,
            cups_o_referencia=contracte,
            tags_extra=tags_extra,
            fields_extra=fields_extra,
        )

    def _fallback_metrics(self, pdf_bytes: bytes, fallback_date: date) -> FacturaMetriques | None:
        amount = parse_euro_amount(extract_text_from_pdf_bytes(pdf_bytes))
        if amount is None:
            return None
        return FacturaMetriques(
            proveidor=self.name,
            tipus=self.tipus,
            numero_factura=f"VEOLIA-{fallback_date.strftime('%Y%m')}",
            data_emissio=fallback_date,
            import_total=amount,
        )

def decode_str(header_value: Any) -> str:
    """Decodifica capçaleres de correu codificades en RFC 2047."""
    if not header_value:
        return ""
    decoded_fragments = decode_header(header_value)
    result = ""
    for frag, enc in decoded_fragments:
        if isinstance(frag, bytes):
            result += frag.decode(enc or "utf-8", errors="replace")
        else:
            result += str(frag)
    return result

    def fetch_recent_emails(
        self,
        sender_filters: list[str] | None = None,
        days_back: int | None = None,
    ) -> list[ParsedEmail]:
        """Cerca i extreu correus recents de la bústia segons filtre de remitent."""
        if not self.is_configured():
            logger.warning("IMAP no està configurat (host, user o password buits). Ometent connexió.")
            return []

        limit_days = days_back or settings.DAYS_BACK
        since_date = (datetime.now() - timedelta(days=limit_days)).strftime("%d-%b-%Y")

        emails: list[ParsedEmail] = []
        try:
            with imaplib.IMAP4_SSL(self.host, self.port) as client:
                client.login(self.user, self.password)
                client.select(self.folder, readonly=not settings.IMAP_MARK_SEEN)

                search_criteria = f'(SINCE "{since_date}")'
                _, data = client.search(None, search_criteria)
                mail_ids = data[0].split()

                logger.info(f"Trobats {len(mail_ids)} correus des de {since_date} a {self.folder}")

                for mail_id in mail_ids:
                    # 1. Optimització: descarregar només capçaleres per filtrar ràpidament
                    if sender_filters:
                        _, hdr_data = client.fetch(mail_id, "(BODY.PEEK[HEADER.FIELDS (FROM SUBJECT)])")
                        if hdr_data and hdr_data[0] and isinstance(hdr_data[0][1], bytes):
                            hdr_msg = email.message_from_bytes(hdr_data[0][1])
                            h_sender = decode_str(hdr_msg.get("From", "")).lower()
                            h_subject = decode_str(hdr_msg.get("Subject", "")).lower()
                            matches_hdr_sender = any(f.lower() in h_sender for f in sender_filters)
                            matches_hdr_subject = any(f.lower() in h_subject for f in sender_filters)
                            is_fwd_or_eml = "fwd:" in h_subject or "reenviat:" in h_subject or "factur" in h_subject
                            if not matches_hdr_sender and not matches_hdr_subject and not is_fwd_or_eml:
                                continue

                    # 2. Descarregar el correu complet només dels correus candidats
                    _, msg_data = client.fetch(mail_id, "(RFC822)")
                    if not msg_data or not msg_data[0]:
                        continue

                    raw_email = msg_data[0][1]
                    if not isinstance(raw_email, bytes):
                        continue

                    msg = email.message_from_bytes(raw_email)
                    sender = decode_str(msg.get("From", ""))
                    subject = decode_str(msg.get("Subject", ""))

                    parsed = self._parse_message(mail_id.decode("utf-8"), msg, sender, subject)

                    # Comprovar si el missatge principal o algun dels missatges .eml imbricats coincideix
                    candidate_messages = [parsed] + parsed.nested_emails
                    for cand in candidate_messages:
                        if sender_filters:
                            matches_sender = any(f.lower() in cand.sender.lower() for f in sender_filters)
                            matches_subject = any(f.lower() in cand.subject.lower() for f in sender_filters)
                            if not matches_sender and not matches_subject:
                                continue
                        emails.append(cand)

        except (imaplib.IMAP4.error, OSError) as e:
            logger.error(f"Error connectant o llegint del servidor IMAP: {e}")

        return emails

    def _parse_message(
        self,
        msg_id: str,
        msg: email.message.Message,
        sender: str,
        subject: str,
    ) -> ParsedEmail:
        body_text_parts: list[str] = []
        attachments: list[EmailAttachment] = []
        nested_emails: list[ParsedEmail] = []

        date_hdr = msg.get("Date")
        date_obj: datetime | None = None
        if date_hdr:
            try:
                date_obj = parsedate_to_datetime(date_hdr)
            except (TypeError, ValueError) as err:
                logger.debug(f"No s'ha pogut parsejar la data {date_hdr}: {err}")

        if msg.is_multipart():
            for part in msg.walk():
                content_type = part.get_content_type()
                content_disposition = str(part.get("Content-Disposition", ""))

                filename = part.get_filename()
                if filename:
                    filename = decode_str(filename)

                # 1. Desempaquetat recursiu de correus adjunts en format .eml o message/rfc822
                if content_type == "message/rfc822" or (filename and filename.lower().endswith(".eml")):
                    try:
                        sub_msg: email.message.Message | None = None
                        if content_type == "message/rfc822":
                            sub_payload = part.get_payload()
                            if (
                                isinstance(sub_payload, list)
                                and len(sub_payload) > 0
                                and isinstance(sub_payload[0], email.message.Message)
                            ):
                                sub_msg = sub_payload[0]
                            elif isinstance(sub_payload, bytes):
                                sub_msg = email.message_from_bytes(sub_payload)
                        elif filename and filename.lower().endswith(".eml"):
                            raw_bytes = part.get_payload(decode=True)
                            if isinstance(raw_bytes, bytes):
                                sub_msg = email.message_from_bytes(raw_bytes)

                        if sub_msg:
                            sub_sender = decode_str(sub_msg.get("From", ""))
                            sub_subject = decode_str(sub_msg.get("Subject", ""))
                            nested_id = f"{msg_id}_{filename or 'nested'}"
                            sub_parsed = self._parse_message(nested_id, sub_msg, sub_sender, sub_subject)
                            nested_emails.append(sub_parsed)
                            logger.info(
                                f"Extret correu .eml adjunt: '{sub_subject}' de '{sub_sender}' ({len(sub_parsed.attachments)} PDFs)"
                            )
                    except (TypeError, ValueError, OSError) as eml_err:
                        logger.warning(f"Error parsejant correu .eml adjunt: {eml_err}")

                # 2. Adjunts normals en format PDF (validació per extensió i magic bytes %PDF)
                elif filename and filename.lower().endswith(".pdf"):
                    payload = part.get_payload(decode=True)
                    if isinstance(payload, bytes) and payload.startswith(b"%PDF"):
                        attachments.append(
                            EmailAttachment(
                                filename=filename,
                                content_type=content_type or "application/pdf",
                                data=payload,
                            )
                        )
                elif "attachment" in content_disposition.lower():
                    # Si està marcat com a attachment però no té nom .pdf, comprovem si és realment un PDF
                    payload = part.get_payload(decode=True)
                    if isinstance(payload, bytes) and payload.startswith(b"%PDF"):
                        attachments.append(
                            EmailAttachment(
                                filename=filename or f"adjunt_{msg_id}.pdf",
                                content_type="application/pdf",
                                data=payload,
                            )
                        )
                elif content_type in ("text/plain", "text/html") and "attachment" not in content_disposition.lower():
                    payload = part.get_payload(decode=True)
                    if isinstance(payload, bytes):
                        body_text_parts.append(payload.decode("utf-8", errors="replace"))
        else:
            payload = msg.get_payload(decode=True)
            if isinstance(payload, bytes):
                body_text_parts.append(payload.decode("utf-8", errors="replace"))

        body_text = "\n".join(body_text_parts)
        # Neteja de tags HTML bàsics si cal
        body_text = re.sub(r"<[^>]+>", " ", body_text)

        return ParsedEmail(
            msg_id=msg_id,
            sender=sender,
            subject=subject,
            date=date_obj,
            body_text=body_text,
            attachments=attachments,
            nested_emails=nested_emails,
        )

import logging

class InfluxMetricsWriter:
    """Gestiona l'escriptura de dades de facturació a InfluxDB v2."""

    def __init__(self) -> None:
        self.url = settings.INFLUXDB_URL
        self.token = settings.INFLUXDB_TOKEN
        self.org = settings.INFLUXDB_ORG
        self.bucket = settings.INFLUXDB_BUCKET

    def is_configured(self) -> bool:
        return bool(settings.INFLUXDB_ENABLED and self.url and self.token and self.bucket)

    def write_factura_metrics(self, metriques: FacturaMetriques) -> bool:
        """Escriu les mètriques d'una factura com a Point de sèrie temporal a InfluxDB."""
        if not self.is_configured():
            logger.info("InfluxDB no està configurat o està desactivat. Ometent escriptura de mètriques.")
            return False

        if settings.DRY_RUN:
            logger.info(
                f"[DRY-RUN] S'escriurien mètriques a InfluxDB per {metriques.proveidor} "
                f"({metriques.numero_factura}) -> Total: {metriques.import_total} €"
            )
            return True

        # Convertim la data de la factura a datetime a migdia UTC per a la sèrie temporal
        dt = datetime.combine(metriques.data_emissio, time(12, 0, 0), tzinfo=UTC)

        point = (
            Point("factures_subministres")
            .tag("proveidor", metriques.proveidor)
            .tag("tipus", metriques.tipus.value)
            .tag("numero_factura", metriques.numero_factura)
            .tag("any", str(metriques.data_emissio.year))
            .tag("mes", f"{metriques.data_emissio.month:02d}")
            .field("import_total", float(metriques.import_total))
            .time(dt)
        )

        if metriques.titular:
            point.tag("titular", metriques.titular)
        if metriques.cups_o_referencia:
            point.tag("referencia", metriques.cups_o_referencia)

        # Camps opcionals monetaris
        if metriques.import_base is not None:
            point.field("import_base", float(metriques.import_base))
        if metriques.import_iva is not None:
            point.field("import_iva", float(metriques.import_iva))
        if metriques.dies_facturats is not None:
            point.field("dies_facturats", int(metriques.dies_facturats))

        # Camps opcionals de consum
        if metriques.consum_m3 is not None:
            point.field("consum_m3", float(metriques.consum_m3))
        if metriques.consum_kwh is not None:
            point.field("consum_kwh", float(metriques.consum_kwh))
        if metriques.consum_kwh_punta is not None:
            point.field("consum_kwh_punta", float(metriques.consum_kwh_punta))
        if metriques.consum_kwh_pla is not None:
            point.field("consum_kwh_pla", float(metriques.consum_kwh_pla))
        if metriques.consum_kwh_vall is not None:
            point.field("consum_kwh_vall", float(metriques.consum_kwh_vall))
        if metriques.potencia_contractada_kw is not None:
            point.field("potencia_contractada_kw", float(metriques.potencia_contractada_kw))

        # Camps telefonia
        if metriques.total_gb_consumits is not None:
            point.field("total_gb_consumits", float(metriques.total_gb_consumits))
        if metriques.num_linies_mobils is not None:
            point.field("num_linies_mobils", int(metriques.num_linies_mobils))

        # Tags i camps addicionals
        for k, v in metriques.tags_extra.items():
            point.tag(k, v)
        for k, num in metriques.fields_extra.items():
            point.field(k, float(num))

        # Càlcul automàtic de ràtios unitaris per a Grafana
        try:
            if metriques.dies_facturats and metriques.dies_facturats > 0:
                point.field("cost_diari_eur", round(float(metriques.import_total) / int(metriques.dies_facturats), 4))
            if metriques.consum_kwh and metriques.consum_kwh > 0:
                point.field("preu_kwh_real", round(float(metriques.import_total) / float(metriques.consum_kwh), 4))
            if metriques.consum_m3 and metriques.consum_m3 > 0:
                point.field("preu_m3_real", round(float(metriques.import_total) / float(metriques.consum_m3), 4))
        except (ValueError, TypeError, ZeroDivisionError) as err:
            logger.debug(f"Error calculant ràtios unitaris: {err}")

        try:
            with InfluxDBClient(url=self.url, token=self.token, org=self.org) as client:
                write_api = client.write_api(write_options=SYNCHRONOUS)
                write_api.write(bucket=self.bucket, org=self.org, record=point)
                logger.info(
                    f"Mètriques enviades a InfluxDB [{self.bucket}] per factura {metriques.numero_factura} "
                    f"de {metriques.proveidor}"
                )
                return True
        except (InfluxDBError, urllib3.exceptions.HTTPError, OSError) as e:
            logger.error(f"Error escrivint a InfluxDB: {e}")
            return False

    def delete_factura_metrics(self, proveidor: str, numero_factura: str | None = None) -> bool:
        """Elimina mètriques d'una factura a InfluxDB per evitar dades errònies a Grafana."""
        if not self.is_configured():
            return False

        if settings.DRY_RUN:
            logger.info(f"[DRY-RUN] S'eliminarien mètriques a InfluxDB per {proveidor} ({numero_factura})")
            return True

        predicate = f'_measurement="factures_subministres" and proveidor="{proveidor.lower()}"'
        if numero_factura:
            predicate += f' and numero_factura="{numero_factura}"'

        try:
            with InfluxDBClient(url=self.url, token=self.token, org=self.org) as client:
                delete_api = client.delete_api()
                delete_api.delete(
                    start="1970-01-01T00:00:00Z",
                    stop="2035-01-01T00:00:00Z",
                    predicate=predicate,
                    bucket=self.bucket,
                    org=self.org,
                )
                logger.info(f"Mètriques eliminades a InfluxDB per {proveidor} ({numero_factura or 'totes'}).")
                return True
        except (InfluxDBError, urllib3.exceptions.HTTPError, OSError) as err:
            logger.warning(f"Error eliminant mètriques d'InfluxDB: {err}")
            return False

    def get_factures_amounts(self) -> dict[str, float]:
        """Recupera un mapa {numero_factura_lower: import_total} d'InfluxDB."""
        if not self.is_configured():
            return {}

        result_map: dict[str, float] = {}
        try:
            with InfluxDBClient(url=self.url, token=self.token, org=self.org, timeout=5000) as client:
                tables = client.query_api().query(
                    f'from(bucket: "{self.bucket}") '
                    "|> range(start: -10y) "
                    '|> filter(fn: (r) => r["_measurement"] == "factures_subministres" and r["_field"] == "import_total") '
                    "|> last()",
                    org=self.org,
                )
                for table in tables:
                    for record in table.records:
                        num = record.values.get("numero_factura")
                        val = record.get_value()
                        if num and val is not None:
                            try:
                                result_map[str(num).strip().lower()] = float(val)
                            except (ValueError, TypeError):
                                continue
        except (InfluxDBError, urllib3.exceptions.HTTPError, OSError) as err:
            logger.debug(f"No s'ha pogut obtenir mapa d'imports d'InfluxDB: {err}")
        return result_map

def run_etl(connector_filter: str | None = None) -> int:
    logger = logging.getLogger("etl_main")
    logger.info("Iniciant cicle ETL de factures de subministraments...")

    all_connectors: list[BaseConnector] = [
        VeoliaConnector(),
        CatgasConnector(),
        BonpreuConnector(),
        DigiConnector(),
    ]

    if connector_filter:
        connectors_to_run = [c for c in all_connectors if c.name.lower() == connector_filter.lower()]
        if not connectors_to_run:
            logger.error(f"Connector '{connector_filter}' no trobat.")
            return 1
    else:
        connectors_to_run = [c for c in all_connectors if c.is_enabled()]

    logger.info(f"Connectors actius a executar: {[c.name for c in connectors_to_run]}")

    all_results: list[FacturaResult] = []
    has_errors = False

    for connector in connectors_to_run:
        logger.info(f"--- Iniciant processament: {connector.name.upper()} ({connector.tipus.value}) ---")
        try:
            results = connector.run(imap_client)
            all_results.extend(results)
            for res in results:
                if res.errors:
                    has_errors = True
        except (OSError, RuntimeError, ValueError) as e:
            logger.error(f"Error crític executant el connector {connector.name}: {e}")
            has_errors = True

    if all_results:
        print_summary_table(all_results)
    else:
        logger.info("No s'ha trobat cap factura nova en el període especificat.")

    logger.info("Cicle ETL finalitzat.")
    return 1 if has_errors else 0

import io
import logging
import re
from datetime import date
from typing import Any

from dateutil import parser as date_parser
from pypdf import PdfReader
from pypdf.errors import PdfReadError

logger = logging.getLogger("pdf_parser")


def extract_text_from_pdf_bytes(pdf_bytes: bytes) -> str:
    """Extreu tot el text pla d'un fitxer PDF en memòria."""
    text_parts: list[str] = []
    try:
        reader = PdfReader(io.BytesIO(pdf_bytes))
        for page in reader.pages:
            extracted = page.extract_text()
            if extracted:
                text_parts.append(extracted)
    except (PdfReadError, OSError, ValueError) as e:
        logger.warning(f"Error extreient text del PDF: {e}")
        return ""
    return "\n".join(text_parts)


def normalize_amount_str(val_str: str) -> float | None:
    """Normalitza una cadena numèrica monetària (ex: '25,00', '9.92', '1.234,56') a float de forma robusta."""
    clean = val_str.strip()
    if not clean:
        return None
    if "," in clean:
        # Format europeu/espanyol estàndard: punts són milers, coma és decimal
        clean = clean.replace(".", "").replace(",", ".")
    elif "." in clean:
        parts = clean.split(".")
        if len(parts) == 2 and len(parts[1]) <= 2:
            # Format float estàndard (ex: 9.92, 25.00): el punt és el separador decimal!
            pass
        elif len(parts) == 2 and len(parts[1]) == 3:
            # Milers sense decimals (ex: 1.000)
            clean = clean.replace(".", "")
        else:
            clean = clean.replace(".", "")

    try:
        return float(clean)
    except ValueError:
        return None


def parse_euro_amount(text: str) -> float | None:
    """Extreu l'import total monetari (ex: 45,67 € o 9.92 €) prioritzant etiquetes de total."""
    # 1. Prioritzar etiquetes explícites de total amb decimals
    total_patterns = [
        r"(?:TOTAL\s+A\s+PAGAR|TOTAL\s+FACTURA|IMPORT\s+TOTAL|IMPORTE\s+TOTAL)[^0-9]{0,40}?(\d{1,4}[.,]\d{2})\s*€",
        r"(\d{1,4}[.,]\d{2})\s*€\s*(?:TOTAL\s+FACTURA|TOTAL\s+A\s+PAGAR)",
        r"TOTAL[:\s]+(\d{1,4}[.,]\d{2})\s*€?",
    ]
    for pat in total_patterns:
        match = re.search(pat, text, re.IGNORECASE | re.DOTALL)
        if match:
            val = normalize_amount_str(match.group(1))
            if val is not None:
                return val

    # 2. Fallback: cercar imports amb decimals explícits seguits de €
    matches = re.findall(r"(\d{1,4}[.,]\d{2})\s*€", text)
    if matches:
        # En moltes factures l'import total és l'últim o el més rellevant amb decimals
        return normalize_amount_str(matches[-1])
    return None


def parse_date_catalan_spanish(date_str: str) -> date | None:
    """Intenta parsejar dates típiques en català o castellà (ex: 15/09/2026, 15 d'octubre de 2026, 09 d'oct. 2026)."""
    months_map = {
        "gener": "01",
        "enero": "01",
        "gen": "01",
        "ene": "01",
        "febrer": "02",
        "febrero": "02",
        "feb": "02",
        "març": "03",
        "marzo": "03",
        "mar": "03",
        "abril": "04",
        "abr": "04",
        "maig": "05",
        "mayo": "05",
        "mai": "05",
        "may": "05",
        "juny": "06",
        "junio": "06",
        "jun": "06",
        "juliol": "07",
        "julio": "07",
        "jul": "07",
        "agost": "08",
        "agosto": "08",
        "ago": "08",
        "setembre": "09",
        "septiembre": "09",
        "set": "09",
        "sep": "09",
        "octubre": "10",
        "oct": "10",
        "novembre": "11",
        "noviembre": "11",
        "nov": "11",
        "desembre": "12",
        "diciembre": "12",
        "des": "12",
        "dic": "12",
    }
    clean = date_str.lower().strip()
    # Netejar connectors en català/castellà com "d'", "d’", "de ", "del "
    clean = re.sub(r"\bd\s*['’]\s*", "", clean)
    clean = re.sub(r"\bde\s+", "", clean)
    clean = re.sub(r"\bdel\s+", "", clean)

    for cat_es, num in months_map.items():
        clean = re.sub(rf"\b{cat_es}\.?\b", num, clean)
    clean = re.sub(r"[^\w\s/.-]", " ", clean)

    try:
        dt = date_parser.parse(clean, dayfirst=True)
        return dt.date()
    except (date_parser.ParserError, ValueError, OverflowError):
        return None


def extract_regex_first(text: str, pattern: str, group: int = 1) -> str | None:
    """Cerca un patró regex i retorna el grup capturat o None."""
    m = re.search(pattern, text, re.IGNORECASE | re.MULTILINE)
    return m.group(group).strip() if m else None


def extract_date_range(text: str, pattern: str) -> tuple[date | None, date | None]:
    """Cerca un patró regex amb dos grups de captura de data i retorna (data_inici, data_fi)."""
    m = re.search(pattern, text, re.IGNORECASE | re.MULTILINE)
    if m and len(m.groups()) >= 2:
        dt_inici = parse_date_catalan_spanish(m.group(1).strip())
        dt_fi = parse_date_catalan_spanish(m.group(2).strip())
        return dt_inici, dt_fi
    return None, None


def extract_float_value(text: str, pattern: str, group: int = 1) -> float | None:
    """Cerca un valor numèric amb regex i el converteix a float de forma robusta."""
    val = extract_regex_first(text, pattern, group)
    if not val:
        return None
    return normalize_amount_str(val)

class StorageManager:
    """Gestiona l'emmagatzematge de factures PDF al volum NAS Samba."""

    def __init__(self, base_path: str | None = None) -> None:
        self.base_path = Path(base_path or settings.NAS_MOUNT_PATH)

    def get_destination_dir(self, tipus: str, proveidor: str, any_str: str) -> Path:
        """Retorna el directori organitzat: base_path / tipus / proveidor / any."""
        dest_dir = self.base_path / tipus.lower() / proveidor.lower() / str(any_str)
        dest_dir.mkdir(parents=True, exist_ok=True)
        return dest_dir

    def calculate_sha256(self, file_content: bytes) -> str:
        """Calcula el hash SHA256 d'un contingut per detectar duplicats exactes."""
        return hashlib.sha256(file_content).hexdigest()

    def exists(self, tipus: str, proveidor: str, any_str: str, filename: str) -> bool:
        """Comprova si el fitxer ja existeix al NAS."""
        target_file = self.get_destination_dir(tipus, proveidor, any_str) / filename
        return target_file.exists()

    def save_pdf(
        self,
        file_bytes: bytes,
        tipus: str,
        proveidor: str,
        any_str: str,
        filename: str,
    ) -> Path:
        """Desa un PDF al NAS i retorna el Path resultant."""
        dest_dir = self.get_destination_dir(tipus, proveidor, any_str)
        target_path = dest_dir / filename

        if settings.DRY_RUN:
            logger.info(f"[DRY-RUN] Es guardaria el fitxer a: {target_path} ({len(file_bytes)} bytes)")
            return target_path

        target_path.write_bytes(file_bytes)
        logger.info(f"Factura desada correctament a: {target_path}")
        return target_path

    def delete_pdf(self, rel_path: str) -> bool:
        """Elimina un fitxer del NAS Samba de forma segura."""
        base_resolved = self.base_path.resolve()
        target = (base_resolved / rel_path).resolve()
        if not target.is_relative_to(base_resolved) or not target.is_file():
            return False
        if settings.DRY_RUN:
            logger.info(f"[DRY-RUN] S'eliminaria el fitxer a: {target}")
            return True
        target.unlink(missing_ok=True)
        logger.info(f"Fitxer eliminat correctament del NAS: {target}")
        return True

class ConnectionManager:
    """Gestiona les connexions WebSocket actives per emetre logs i notificacions de versió."""

    def __init__(self) -> None:
        self.active_connections: list[WebSocket] = []

    async def connect(self, websocket: WebSocket) -> None:
        await websocket.accept()
        self.active_connections.append(websocket)

    def disconnect(self, websocket: WebSocket) -> None:
        if websocket in self.active_connections:
            self.active_connections.remove(websocket)

    async def broadcast_log(self, formatted_line: str) -> None:
        for ws in list(self.active_connections):
            try:
                await ws.send_json({"type": "log", "line": formatted_line})
            except (WebSocketDisconnect, ConnectionResetError, OSError):
                self.disconnect(ws)

    async def broadcast_version(self, version: str) -> None:
        for ws in list(self.active_connections):
            try:
                await ws.send_json({"type": "version", "version": version})
            except (WebSocketDisconnect, ConnectionResetError, OSError):
                self.disconnect(ws)

    async def broadcast_status(self, payload: dict[str, Any]) -> None:
        for ws in list(self.active_connections):
            try:
                await ws.send_json({"type": "status", **payload})
            except (WebSocketDisconnect, ConnectionResetError, OSError):
                self.disconnect(ws)

    async def broadcast_progress(self, payload: dict[str, Any]) -> None:
        for ws in list(self.active_connections):
            try:
                await ws.send_json({"type": "progress", **payload})
            except (WebSocketDisconnect, ConnectionResetError, OSError):
                self.disconnect(ws)

def auto_annotate_k8s_pod() -> None:
    """Anota automàticament el Pod actual a Kubernetes amb la versió SemVer de l'aplicació."""
    pod_name = os.getenv("HOSTNAME")
    if not pod_name:
        return

    token_path = Path("/var/run/secrets/kubernetes.io/serviceaccount/token")
    if not token_path.exists():
        logger.debug("Execució fora de Kubernetes. Ometent auto-anotació del Pod.")
        return

    token = token_path.read_text().strip()
    ns_path = Path("/var/run/secrets/kubernetes.io/serviceaccount/namespace")
    namespace = ns_path.read_text().strip() if ns_path.exists() else "subministres"
    ca_cert = "/var/run/secrets/kubernetes.io/serviceaccount/ca.crt"

    url = f"https://kubernetes.default.svc/api/v1/namespaces/{namespace}/pods/{pod_name}"
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/strategic-merge-patch+json",
    }
    payload = {"metadata": {"annotations": {"app.kubernetes.io/version": f"v{settings.APP_VERSION}"}}}

    try:
        resp = requests.patch(
            url,
            json=payload,
            headers=headers,
            verify=ca_cert if Path(ca_cert).exists() else False,
            timeout=5,
        )
        if resp.status_code in (200, 201):
            logger.info(f"✅ Pod {pod_name} auto-anotat a Kubernetes amb versió: v{settings.APP_VERSION}")
        else:
            logger.warning(f"No s'ha pogut auto-anotar el Pod (HTTP {resp.status_code}): {resp.text}")
    except (OSError, RuntimeError, requests.exceptions.RequestException) as err:
        logger.warning(f"Error intentant auto-anotar el Pod a Kubernetes: {err}")

def list_nas_invoices() -> list[dict[str, Any]]:
    """Explora el directori del NAS Samba per llistar totes les factures existents amb mes i import."""
    root = Path(settings.NAS_MOUNT_PATH)
    if not root.exists():
        return []

    amounts_map = influx_writer.get_factures_amounts()
    invoices: list[dict[str, Any]] = []

    for pdf_path in root.glob("**/*.pdf"):
        try:
            stat = pdf_path.stat()
            rel_path = pdf_path.relative_to(root)
            parts = rel_path.parts  # [tipus, proveidor, any, filename]
            tipus = parts[0] if len(parts) > 0 else "general"
            proveidor = parts[1] if len(parts) > 1 else "desconegut"
            any_str = parts[2] if len(parts) > 2 else ""

            # 1. Deducció del Mes o Trimestre (ex: 2026_10_catgas... -> 10, 2026_T3_veolia... -> T3)
            fn_parts = pdf_path.stem.split("_")
            mes_code = "01"
            if len(fn_parts) >= 2 and fn_parts[1].isdigit():
                mes_code = fn_parts[1].zfill(2)
            elif len(fn_parts) >= 2 and fn_parts[1].upper() in ("T1", "T2", "T3", "T4", "Q1", "Q2", "Q3", "Q4"):
                mes_code = fn_parts[1].upper().replace("Q", "T")
            elif len(fn_parts) >= 2 and "-" in fn_parts[1]:
                # Si és bimensual/trimestral (ex: 09-10), agafem el primer mes del rang
                sub_p = fn_parts[1].split("-")
                if sub_p[0].isdigit():
                    mes_code = sub_p[0].zfill(2)
            else:
                mes_code = datetime.fromtimestamp(stat.st_mtime).strftime("%m")

            # Mapeig especial per a factures d'aigua trimestrals (Veolia / Agbar)
            if proveidor.lower() == "veolia":
                if mes_code in ("01", "02", "03", "T1"):
                    mes_label = "1r Trim (T1)"
                elif mes_code in ("04", "05", "06", "T2"):
                    mes_label = "2n Trim (T2)"
                elif mes_code in ("07", "08", "09", "10", "T3"):
                    mes_label = "3r Trim (T3)"
                elif mes_code in ("11", "12", "T4"):
                    mes_label = "4t Trim (T4)"
                else:
                    mes_label = f"{MESOS_CATALANS.get(mes_code, mes_code)} ({mes_code})"
            else:
                mes_label = f"{MESOS_CATALANS.get(mes_code, mes_code)} ({mes_code})"

            # Deducció de data ISO (YYYY-MM-DD) per al filtratge per rang a la web
            clean_year = (
                any_str
                if any_str.isdigit() and len(any_str) == 4
                else datetime.fromtimestamp(stat.st_mtime).strftime("%Y")
            )
            date_month = "01"
            if mes_code.isdigit():
                date_month = mes_code.zfill(2)
            elif mes_code == "T1":
                date_month = "03"
            elif mes_code == "T2":
                date_month = "06"
            elif mes_code == "T3":
                date_month = "09"
            elif mes_code == "T4":
                date_month = "12"
            date_iso = f"{clean_year}-{date_month}-01"

            # 2. Deducció de l'Import
            num_clean = pdf_path.stem.lower()
            for p in ["catgas", "veolia", "bonpreu", "digi"]:
                if f"_{p}_" in num_clean:
                    num_clean = num_clean.split(f"_{p}_", 1)[1]
                    break

            import_val = amounts_map.get(num_clean) or amounts_map.get(pdf_path.stem.lower())
            if import_val is None:
                cached = _pdf_amount_cache.get(str(rel_path))
                if cached and cached[0] == stat.st_mtime:
                    import_val = cached[1]
                else:
                    try:
                        pdf_data = pdf_path.read_bytes()
                        conn = CONNECTORS_MAP.get(proveidor.lower())
                        metriques = conn.extract_metrics_from_pdf(pdf_data) if conn else None
                        if metriques and metriques.import_total is not None:
                            import_val = metriques.import_total
                        else:
                            txt = extract_text_from_pdf_bytes(pdf_data)
                            import_val = parse_euro_amount(txt)
                        if import_val is not None:
                            _pdf_amount_cache[str(rel_path)] = (stat.st_mtime, import_val)
                    except (OSError, RuntimeError, ValueError):
                        pass

            if import_val is not None:
                import_str = f"{import_val:.2f} €"
            else:
                import_str = "- €"
                import_val = 0.0

            invoices.append(
                {
                    "filename": pdf_path.name,
                    "tipus": tipus,
                    "proveidor": proveidor,
                    "any": any_str,
                    "mes": mes_code,
                    "mes_label": mes_label,
                    "date_iso": date_iso,
                    "import_val": import_val,
                    "import_str": import_str,
                    "size_kb": round(stat.st_size / 1024, 1),
                    "modified": datetime.fromtimestamp(stat.st_mtime).strftime("%Y-%m-%d %H:%M"),
                    "rel_path": str(rel_path).replace("\\", "/"),
                }
            )
        except OSError:
            continue

    # Ordenar per data de modificació descendent
    invoices.sort(key=lambda x: x["modified"], reverse=True)
    return invoices


def run_sync_task(
    connector_name: str | None = None,
    mode: str = "normal",
    limit: int | None = None,
    source: str = "auto",
) -> None:
    """Tasca d'execució d'ETL amb emissió de progrés (X/N) i diagnòstic en temps real."""
    source_label = {"email": "📧 Correu IMAP", "portal": "🌐 Portal Web", "auto": "⚡ Híbrid"}.get(source, source)
    execution_state["is_running"] = True
    execution_state["active_connector"] = f"{(connector_name or 'Tots').upper()} ({source_label})"
    execution_state["last_status"] = "Sincronitzant..."

    def _emit_progress(data: dict[str, Any]) -> None:
        if main_loop and main_loop.is_running():
            asyncio.run_coroutine_threadsafe(ws_manager.broadcast_progress(data), main_loop)

    connectors_map = get_all_connectors_map()
    if connector_name and connector_name.lower() in connectors_map:
        to_run = [connectors_map[connector_name.lower()]]
    else:
        to_run = [c for c in connectors_map.values() if c.is_enabled()]

    results: list[FacturaResult] = []
    error_messages: list[str] = []

    for conn in to_run:
        conn.progress_callback = _emit_progress
        try:
            _emit_progress(
                {
                    "connector": conn.name.upper(),
                    "step": f"Iniciant extracció ({source_label})...",
                    "current": 0,
                    "total": 0,
                    "percent": 0,
                    "status": "running",
                }
            )
            res = conn.run(imap_client, mode=mode, limit=limit, source=source)
            results.extend(res)
            for r in res:
                if r.errors:
                    error_messages.extend(r.errors)
        except (OSError, RuntimeError, ValueError) as err:
            err_str = f"{conn.name.upper()}: {err}"
            logger.error(f"❌ Error executant {err_str}")
            error_messages.append(err_str)
        finally:
            conn.progress_callback = None

    noves = sum(1 for r in results if r.nou_descarregat)
    existents = len(results) - noves

    execution_state["is_running"] = False
    execution_state["last_run"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    execution_state["last_status"] = "Error" if error_messages else "Completat"
    execution_state["last_results"] = [r.model_dump() for r in results]
    execution_state["active_connector"] = None

    if error_messages:
        summary_msg = f"❌ Finalitzat amb errors ({source_label}): {'; '.join(error_messages)}"
        status_code = "error"
    elif len(results) == 0:
        summary_msg = (
            f"⚠️ Procés finalitzat ({source_label}): 0 factures obtingudes. "
            "Revisa la consola de logs inferior per veure el detall."
        )
        status_code = "warning"
    else:
        summary_msg = (
            f"✅ Completat ({source_label}): {len(results)} factures processades "
            f"({noves} noves desades al NAS, {existents} ja existents)."
        )
        status_code = "ok"

    logger.info(summary_msg)
    if main_loop and main_loop.is_running():
        asyncio.run_coroutine_threadsafe(
            ws_manager.broadcast_status({"status": status_code, "message": summary_msg, "new_count": noves}),
            main_loop,
        )

            await websocket.send_json({"type": "log", "line": line})

def detect_provider_from_content(pdf_bytes: bytes, filename: str) -> str:
    """Detecta el proveïdor analitzant el nom de fitxer o el text del PDF."""
    text = extract_text_from_pdf_bytes(pdf_bytes) or ""
    combined = (filename + " " + text).lower()

    if "bonpreu" in combined or "esclat" in combined:
        return "bonpreu"
    if "digi" in combined or "digimobil" in combined:
        return "digi"
    if "veolia" in combined or "agbar" in combined or "aigües" in combined or "aigues" in combined:
        return "veolia"
    if "catgas" in combined:
        return "catgas"
    return "bonpreu"


def extract_pdfs_from_eml(eml_bytes: bytes) -> list[tuple[bytes, str, str | None]]:
    """Extreu fitxers PDF d'un correu .eml, retornant (pdf_bytes, filename, provider_hint)."""
    results: list[tuple[bytes, str, str | None]] = []
    try:
        msg = email.message_from_bytes(eml_bytes)
        sender = decode_str(msg.get("From", ""))
        subject = decode_str(msg.get("Subject", ""))
        combined_meta = f"{sender} {subject}".lower()

        provider_hint: str | None = None
        if "bonpreu" in combined_meta:
            provider_hint = "bonpreu"
        elif "digi" in combined_meta:
            provider_hint = "digi"
        elif "veolia" in combined_meta or "agbar" in combined_meta:
            provider_hint = "veolia"
        elif "catgas" in combined_meta:
            provider_hint = "catgas"

        for part in msg.walk():
            ct = part.get_content_type()
            cd = str(part.get("Content-Disposition", "")).lower()
            fname = part.get_filename()
            if fname:
                fname = decode_str(fname)

            is_pdf_name = bool(fname and fname.lower().endswith(".pdf"))
            is_image = bool(
                fname and fname.lower().endswith((".png", ".jpg", ".jpeg", ".gif", ".svg", ".ico", ".webp"))
            )
            if is_image:
                continue

            if is_pdf_name or "attachment" in cd:
                payload = part.get_payload(decode=True)
                if isinstance(payload, bytes) and payload.startswith(b"%PDF"):
                    results.append((payload, fname or "adjunt.pdf", provider_hint))
            elif ct == "message/rfc822" or (fname and fname.lower().endswith(".eml")):
                sub_payload = part.get_payload(decode=True)
                if isinstance(sub_payload, bytes):
                    results.extend(extract_pdfs_from_eml(sub_payload))
    except (TypeError, ValueError, OSError) as e:
        logger.warning(f"Error parsejant fitxer EML: {e}")
    return results


MAX_UPLOAD_BYTES = 50 * 1024 * 1024  # Límit màxim de 50 MB per seguretat


@app.post("/api/upload")
async def upload_invoice_files(
    file: Annotated[UploadFile, File()],
    proveidor: Annotated[str, Form()] = "auto",
):
    """Pujada manual de factures (Dropzone): admet PDF, EML o ZIP (amb múltiples PDFs/EMLs)."""
    if not file.filename:
        raise HTTPException(status_code=400, detail="Nom de fitxer no proporcionat.")

    fname_lower = file.filename.lower()
    if not fname_lower.endswith((".pdf", ".eml", ".zip")):
        raise HTTPException(
            status_code=400,
            detail="Format no admès. Només es permeten fitxers .pdf, .eml o .zip.",
        )

    content = await file.read(MAX_UPLOAD_BYTES + 1)
    if len(content) == 0:
        raise HTTPException(status_code=400, detail="El fitxer és buit.")
    if len(content) > MAX_UPLOAD_BYTES:
        raise HTTPException(status_code=413, detail="El fitxer supera el límit màxim de 50 MB.")

    connectors = get_all_connectors_map()

    # Recollir tots els PDFs a processar: llista de (pdf_bytes, filename, provider_hint)
    pdf_queue: list[tuple[bytes, str, str | None]] = []

    if fname_lower.endswith(".zip"):
        logger.info(f"Desempaquetant fitxer ZIP: {file.filename}")
        try:
            with zipfile.ZipFile(io.BytesIO(content)) as z:
                for zinfo in z.infolist():
                    if zinfo.is_dir() or zinfo.file_size > MAX_UPLOAD_BYTES:
                        continue
                    inner_name = zinfo.filename.lower()
                    if inner_name.endswith(".pdf"):
                        pdf_data = z.read(zinfo)
                        if pdf_data.startswith(b"%PDF"):
                            pdf_queue.append((pdf_data, Path(zinfo.filename).name, None))
                    elif inner_name.endswith(".eml"):
                        eml_data = z.read(zinfo)
                        pdf_queue.extend(extract_pdfs_from_eml(eml_data))
        except zipfile.BadZipFile:
            raise HTTPException(status_code=400, detail="El fitxer ZIP no és vàlid o està malmès.")
    elif fname_lower.endswith(".eml"):
        logger.info(f"Processant fitxer EML: {file.filename}")
        pdf_queue.extend(extract_pdfs_from_eml(content))
    else:
        # PDF individual
        pdf_queue.append((content, file.filename, None))

    if not pdf_queue:
        return HTMLResponse(
            '<div class="p-4 text-sm text-yellow-800 rounded-lg bg-yellow-50 dark:bg-gray-800 dark:text-yellow-300">'
            "⚠️ No s'ha trobat cap factura en PDF vàlida dins del fitxer pujat.</div>"
        )

    # Processar cada PDF de la cua
    total_processats = 0
    nous_descarregats = 0
    ja_existents = 0
    total_import = 0.0

    for pdf_bytes, filename, hint in pdf_queue:
        # Determinar el proveïdor (hint, manual o autodetectat)
        target_prov = proveidor.lower()
        if target_prov == "auto" or target_prov not in connectors:
            target_prov = hint or detect_provider_from_content(pdf_bytes, filename)

        conn = connectors.get(target_prov, connectors["bonpreu"])
        metriques = conn.extract_metrics_from_pdf(pdf_bytes)

        if not metriques:
            metriques = FacturaMetriques(
                proveidor=conn.name,
                tipus=conn.tipus,
                numero_factura=f"MANUAL-{filename.replace('.pdf', '')}",
                data_emissio=date.today(),
                import_total=0.0,
            )

        res = conn.save_and_ingest(pdf_bytes, metriques, origen="manual_web_upload")
        total_processats += 1
        if res.nou_descarregat:
            nous_descarregats += 1
        else:
            ja_existents += 1

        total_import += metriques.import_total

    summary = (
        f"🎉 S'han processat amb èxit <strong>{total_processats}</strong> factures "
        f"(<strong>{nous_descarregats}</strong> noves desades al NAS, {ja_existents} ja existents omeses). "
        f"Import total registrat a InfluxDB: <strong>{total_import:.2f} €</strong>."
    )
    return HTMLResponse(
        f'<div class="p-4 text-sm text-emerald-800 rounded-lg bg-emerald-50 dark:bg-gray-800 dark:text-emerald-400">'
        f"✅ {summary}</div>"
    )

@app.delete("/api/invoice/{rel_path:path}")
async def delete_invoice(rel_path: str):
    """Elimina una factura del NAS Samba i suprimeix les mètriques d'InfluxDB per evitar dades errònies a Grafana."""
    parts = Path(rel_path).parts
    proveidor = parts[1] if len(parts) > 1 else ""

    # Deduïm el número de factura preservant majúscules/minúscules
    stem = Path(rel_path).stem
    numero_factura = stem
    for p in ["bonpreu", "digi", "veolia", "catgas"]:
        match = re.split(rf"_{p}_", stem, maxsplit=1, flags=re.IGNORECASE)
        if len(match) == 2:
            numero_factura = match[1]
            break

    # 1. Eliminar del NAS i de la memòria cau
    deleted_nas = storage.delete_pdf(rel_path)
    _pdf_amount_cache.pop(rel_path, None)
    _pdf_amount_cache.pop(rel_path.replace("/", "\\"), None)

    # 2. Eliminar d'InfluxDB
    deleted_influx = False
    if proveidor:
        deleted_influx = influx_writer.delete_factura_metrics(proveidor, numero_factura)

    logger.info(
        f"Factura '{rel_path}' eliminada: NAS={deleted_nas}, InfluxDB={deleted_influx} (Proveïdor: {proveidor}, Factura: {numero_factura})"
    )
    return HTMLResponse(content="", status_code=200)

  <script>
    // Inicialitzar icones Lucide
    lucide.createIcons();

    // -------------------------------------------------------------
    // Control de Mida de Font / Zoom (ideal per a monitors 32" o 4K)
    // -------------------------------------------------------------
    const fontSizes = [14, 16, 18, 20, 22, 24];
    let currentFontIndex = 1; // 16px per defecte

    function initFontSize() {
      const saved = localStorage.getItem('subministres_font_size');
      if (saved) {
        const parsed = parseInt(saved, 10);
        const idx = fontSizes.indexOf(parsed);
        if (idx !== -1) currentFontIndex = idx;
      }
      applyFontSize();
    }

    function changeFontSize(delta) {
      currentFontIndex = Math.max(0, Math.min(fontSizes.length - 1, currentFontIndex + delta));
      localStorage.setItem('subministres_font_size', fontSizes[currentFontIndex]);
      applyFontSize();
    }

    function applyFontSize() {
      const size = fontSizes[currentFontIndex];
      document.documentElement.style.fontSize = size + 'px';
      const label = document.querySelector('#font-size-label');
      if (label) {
        const pct = Math.round((size / 16) * 100);
        label.textContent = pct + '%';
      }
    }

    // Inicialitzar mida de font
    initFontSize();

    // -------------------------------------------------------------
    // Modals
    // -------------------------------------------------------------
    function openPdfModal(url, title) {
      const modalTitle = document.querySelector('#pdf-modal-title');
      const frame = document.querySelector('#pdf-frame');
      const modal = document.querySelector('#pdf-modal');
      if (modalTitle && frame && modal) {
        modalTitle.textContent = title;
        frame.setAttribute('src', url);
        modal.classList.remove('hidden');
      }
    }

    function closePdfModal() {
      const frame = document.querySelector('#pdf-frame');
      const modal = document.querySelector('#pdf-modal');
      if (modal && frame) {
        modal.classList.add('hidden');
        frame.setAttribute('src', '');
      }
    }

    function filterInvoices() {
      const q = document.querySelector('#search-input')?.value.toLowerCase().trim() || '';
      const prov = document.querySelector('#provider-filter')?.value.toLowerCase() || 'all';
      const dFrom = document.querySelector('#date-from')?.value || '';
      const dTo = document.querySelector('#date-to')?.value || '';
      const rows = document.querySelectorAll('.invoice-row');
      let visibleCount = 0;

      // Seguiment de proveïdors ja mostrats si el mode "només darreres" està actiu
      const seenProviders = new Set();

      rows.forEach(r => {
        const text = r.textContent?.toLowerCase() || '';
        const rProv = r.getAttribute('data-provider') || r.querySelector('td:first-child')?.textContent?.toLowerCase().trim() || '';
        const rDate = r.getAttribute('data-date') || '';

        const matchesText = !q || text.includes(q);
        const matchesProv = (prov === 'all') || (rProv === prov);

        let matchesDate = true;
        if (dFrom && rDate) {
          matchesDate = matchesDate && (rDate >= dFrom);
        }
        if (dTo && rDate) {
          matchesDate = matchesDate && (rDate <= dTo);
        }

        let isLatestAllowed = true;
        if (onlyLatestActive) {
          if (seenProviders.has(rProv)) {
            isLatestAllowed = false;
          }
        }

        if (matchesText && matchesProv && matchesDate && isLatestAllowed) {
          r.style.display = '';
          visibleCount++;
          if (onlyLatestActive) {
            seenProviders.add(rProv);
          }
        } else {
          r.style.display = 'none';
        }
      });

      const countElem = document.querySelector('#visible-count');
      if (countElem) {
        countElem.textContent = visibleCount;
      }
    }

    const sortStates = {};

    function sortTable(columnIndex) {
      const table = document.querySelector("#invoices-table");
      if (!table) return;
      const tbody = table.querySelector("tbody");
      if (!tbody) return;
      const rows = Array.from(tbody.querySelectorAll("tr.invoice-row"));
      if (rows.length === 0) return;

      const isAsc = sortStates[columnIndex] !== 'asc';
      sortStates[columnIndex] = isAsc ? 'asc' : 'desc';

      // Actualitzar icones de tots els headers
      table.querySelectorAll(".sort-icon").forEach((el, idx) => {
        if (idx === columnIndex) {
          el.textContent = isAsc ? '▲' : '▼';
          el.className = 'sort-icon text-[9px] text-emerald-400 font-bold';
        } else {
          el.textContent = '↕';
          el.className = 'sort-icon text-[9px] text-slate-500';
        }
      });

      rows.sort((a, b) => {
        const valA = a.children[columnIndex]?.textContent?.trim() || '';
        const valB = b.children[columnIndex]?.textContent?.trim() || '';

        // Comprovar si són imports o mides numèriques (ex: 70.57 € o 361.3 KB)
        const cleanA = valA.replace('€', '').replace('KB', '').trim();
        const cleanB = valB.replace('€', '').replace('KB', '').trim();
        const numA = parseFloat(cleanA);
        const numB = parseFloat(cleanB);

        if (!isNaN(numA) && !isNaN(numB) && cleanA.length > 0 && cleanB.length > 0 && !isNaN(cleanA.charAt(0)) && !isNaN(cleanB.charAt(0))) {
          return isAsc ? numA - numB : numB - numA;
        }

        return isAsc ? valA.localeCompare(valB) : valB.localeCompare(valA);
      });

      rows.forEach(r => tbody.appendChild(r));
    }

    const CURRENT_VERSION = 'v{{ settings.APP_VERSION }}';

    let wsReconnectTimeout = null;

    function initWebSocketLogs() {
      const dot = document.querySelector('#ws-status-dot');
      const text = document.querySelector('#ws-status-text');
      const container = document.querySelector('#logs-container');

      const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
      const wsUrl = `${protocol}//${window.location.host}/ws/logs`;

      try {
        logsSocket = new WebSocket(wsUrl);

        logsSocket.onopen = function() {
          clearTimeout(wsReconnectTimeout);
          if (dot) {
            dot.className = 'w-2 h-2 rounded-full bg-emerald-400 animate-pulse';
          }
          if (text) {
            text.textContent = 'En directe (WebSocket)';
            text.className = 'text-[11px] text-emerald-400 font-mono';
          }
        };

        logsSocket.onmessage = function(event) {
          try {
            const data = JSON.parse(event.data);

            if (data.type === 'ping') return;

            // A) Progrés pas a pas en temps real (ex: 3/40 o avís de timeout)
            if (data.type === 'progress') {
              const notif = document.querySelector('#notification-area');
              if (notif && !document.querySelector('#live-progress-bar')) {
                while (notif.firstChild) notif.removeChild(notif.firstChild);
                const box = document.createElement('div');
                box.className = 'p-4 mb-4 rounded-xl bg-slate-900 border border-blue-700/60 shadow-lg space-y-2';

                const header = document.createElement('div');
                header.className = 'flex items-center justify-between text-xs font-semibold text-blue-300';
                const titleEl = document.createElement('span');
                titleEl.id = 'live-progress-title';
                titleEl.textContent = `⏳ Executant ${data.connector || 'ETL'}...`;
                const counterEl = document.createElement('span');
                counterEl.id = 'live-progress-counter';
                counterEl.className = 'font-mono text-emerald-400';
                header.appendChild(titleEl);
                header.appendChild(counterEl);

                const track = document.createElement('div');
                track.className = 'w-full bg-slate-800 rounded-full h-2 overflow-hidden';
                const barEl = document.createElement('div');
                barEl.id = 'live-progress-bar';
                barEl.className = 'bg-emerald-500 h-2 rounded-full transition-all duration-300';
                barEl.style.width = '10%';
                track.appendChild(barEl);

                const stepEl = document.createElement('p');
                stepEl.id = 'live-progress-step';
                stepEl.className = 'text-xs text-slate-300 font-mono truncate';

                box.appendChild(header);
                box.appendChild(track);
                box.appendChild(stepEl);
                notif.appendChild(box);
              }
              const titleEl = document.querySelector('#live-progress-title');
              const counterEl = document.querySelector('#live-progress-counter');
              const barEl = document.querySelector('#live-progress-bar');
              const stepEl = document.querySelector('#live-progress-step');

              if (titleEl && data.connector) titleEl.textContent = `⏳ Processant ${data.connector}...`;
              if (stepEl && data.step) stepEl.textContent = data.step;
              if (counterEl) {
                counterEl.textContent = data.total > 0 ? `${data.current} / ${data.total} (${data.percent}%)` : 'En curs...';
              }
              if (barEl) {
                barEl.style.width = (data.total > 0 ? Math.max(8, data.percent) : 25) + '%';
                if (data.status === 'warning') barEl.className = 'bg-amber-500 h-2 rounded-full transition-all duration-300';
                else if (data.status === 'error') barEl.className = 'bg-red-500 h-2 rounded-full transition-all duration-300';
                else barEl.className = 'bg-emerald-500 h-2 rounded-full transition-all duration-300';
              }
            }

            // B) Resultat Final d'Execució (OK / Warning / Error)
            if (data.type === 'status' && data.message) {
              const notif = document.querySelector('#notification-area');
              if (notif) {
                while (notif.firstChild) notif.removeChild(notif.firstChild);
                const statusBox = document.createElement('div');
                let styles = 'text-blue-300 bg-blue-950/60 border-blue-700/50';
                if (data.status === 'error') styles = 'text-red-300 bg-red-950/70 border-red-700/60';
                else if (data.status === 'warning') styles = 'text-amber-300 bg-amber-950/70 border-amber-700/60';
                else if (data.status === 'ok') styles = 'text-emerald-300 bg-emerald-950/70 border-emerald-700/60';

                statusBox.className = `p-4 mb-4 text-sm rounded-xl border ${styles} font-medium shadow`;
                statusBox.textContent = data.message;
                notif.appendChild(statusBox);
                if (data.new_count && data.new_count > 0) {
                  setTimeout(() => window.location.reload(), 2000);
                }
              }
            }

            // 1. Sincronització de Versió del Servidor
            if (data.type === 'version' && data.version) {
              if (CURRENT_VERSION && data.version !== CURRENT_VERSION) {
                console.log(`Versió del servidor (${data.version}) diferent de la pàgina (${CURRENT_VERSION}). Recarregant...`);
                const banner = document.querySelector('#version-upgrade-banner');
                const bannerText = document.querySelector('#version-upgrade-text');
                if (banner && bannerText) {
                  bannerText.textContent = `🎉 Nova versió del servidor detectada (${data.version}). Sincronitzant pàgina...`;
                  banner.classList.remove('hidden');
                }
                setTimeout(() => {
                  window.location.reload();
                }, 1800);
              }
            }

            // 2. Recepció de Logs en Viu
            if (data.type === 'log' && data.line && container) {
              const entry = document.createElement('div');
              let color = 'text-gray-300';
              if (data.line.includes('ERROR')) color = 'text-red-400 font-semibold';
              else if (data.line.includes('WARNING')) color = 'text-yellow-300';
              else if (data.line.includes('OK') || data.line.includes('correctament') || data.line.includes('desada')) color = 'text-emerald-400';

              entry.className = `${color} py-0.5`;
              entry.textContent = data.line;
              container.appendChild(entry);
              container.scrollTop = container.scrollHeight;
            }
          } catch (e) {
            console.error('Error parsejant missatge WS:', e);
          }
        };

        logsSocket.onclose = function() {
          if (dot) {
            dot.className = 'w-2 h-2 rounded-full bg-amber-500 animate-ping';
          }
          if (text) {
            text.textContent = 'Reconnectant WebSocket...';
            text.className = 'text-[11px] text-amber-400 font-mono';
          }
          // Intentar reconnectar cada 2.5 segons
          clearTimeout(wsReconnectTimeout);
          wsReconnectTimeout = setTimeout(initWebSocketLogs, 2500);
        };

        logsSocket.onerror = function() {
          if (logsSocket) logsSocket.close();
        };

      } catch (err) {
        console.error('No s\'ha pogut establir connexió WebSocket:', err);
      }
    }

    // Inicialitzar el WebSocket al carregar la pàgina
    initWebSocketLogs();
  </script>
